            price: 1
        }
    }
])

    {
        $limit: 5
    }
])
db.productsAgg.aggregate([
    {
        $skip: 3
    }
])
db.productsAgg.aggregate([
    {
        $addFields: {
            totalStockValue: { $multiply: ["$quantity", "$price"] }
        }
    }
])
db.productsAgg.aggregate([
    {
        $set: {
            availability: "In Stock"
        }
    }
])
db.productsAgg.aggregate([
    {
        $project: {
            _id: 0,
            productName: 1,
            price: 1
        }
    }
])
db.productsAgg.aggregate([
    {
        $sample: {
            size: 3
        }
    }
])
db.productsAgg.aggregate([
    {
        $match: {
            price: { $gte: 5000 }
        }
    },
    {
        $out: "premiumProducts"
    }
])
db.productsAgg.aggregate([
    {
        $facet: {
            totalProducts: [
                { $count: "count" }
            ],

            averagePrice: [
                {
                    $group: {
                        _id: null,
                        average: { $avg: "$price" }
                    }
                }
            ]
        }
    }
])
