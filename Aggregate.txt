db.studentsAgg.aggregate([
    {
        $project: {
            _id: 0,
            name: 1,
            department: 1,
            cgpa: 1
        }
    }
])



db.studentsAgg.aggregate([
    {
        $limit: 5
    }
])



db.studentsAgg.aggregate([
    {
        $skip: 3
    }
])



db.studentsAgg.aggregate([
    {
        $addFields: {
            cgpaPercentage: { $multiply: ["$cgpa", 10] }
        }
    }
])



db.studentsAgg.aggregate([
    {
        $set: {
            studentType: "Full-Time"
        }
    }
])



db.studentsAgg.aggregate([
    {
        $project: {
            _id: 0,
            name: 1,
            cgpa: 1
        }
    }
])



db.studentsAgg.aggregate([
    {
        $sample: {
            size: 3
        }
    }
])



db.studentsAgg.aggregate([
    {
        $match: {
            cgpa: { $gte: 8.5 }
        }
    },
    {
        $out: "topCGPAStudents"
    }
])



db.studentsAgg.aggregate([
    {
        $facet: {
            totalStudents: [
                { $count: "count" }
            ],

            averageCGPA: [
                {
                    $group: {
                        _id: null,
                        average: { $avg: "$cgpa" }
                    }
                }
            ]
        }
    }
])
